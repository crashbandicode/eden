cmake_minimum_required(VERSION 3.22)
project(nx2_sampler_probe LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(nx2_sampler_probe sampler_test.cpp ../../../src/yuzu_cmd/nx2_sampler.cpp)
target_include_directories(nx2_sampler_probe PRIVATE ../../../src)
target_compile_features(nx2_sampler_probe PRIVATE cxx_std_20)
target_link_libraries(nx2_sampler_probe PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(nx2_sampler_probe PRIVATE /W4 /WX /EHsc)
    target_compile_definitions(nx2_sampler_probe PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(nx2_sampler_probe PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME nx2_sampler_probe COMMAND nx2_sampler_probe)
set_tests_properties(nx2_sampler_probe PROPERTIES TIMEOUT 20)
